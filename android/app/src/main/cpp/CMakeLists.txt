cmake_minimum_required(VERSION 3.22)
project(omnivae_local LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set_property(GLOBAL PROPERTY JOB_POOLS compile_pool=2 link_pool=1)
set(CMAKE_JOB_POOL_COMPILE compile_pool)
set(CMAKE_JOB_POOL_LINK link_pool)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_LLAMAFILE OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_COMMON OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(LLAMA_CURL OFF CACHE BOOL "" FORCE)
# Inference kernels must remain optimized even in a debuggable APK.
add_compile_options(-O3)
include(FetchContent)
FetchContent_Declare(llama URL https://codeload.github.com/ggml-org/llama.cpp/tar.gz/abeada335e2e78bd3fe63febafab7e900ce75810 URL_HASH SHA256=9d5bfc27b3c3c3fc9fef962587381f071c44c4b216e2c32a12b69b7d20165b32 DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
FetchContent_MakeAvailable(llama)
add_library(omnivae_gguf SHARED offline.cpp)
target_link_libraries(omnivae_gguf PRIVATE llama android log)
